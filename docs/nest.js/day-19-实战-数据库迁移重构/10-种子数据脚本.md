# 种子数据脚本

## 配置 package.json

```json
{
  "prisma": {
    "seed": "ts-node prisma/seed.ts"
  }
}
```

或者使用 `tsx`（更快的 TypeScript 执行器）：

```json
{
  "prisma": {
    "seed": "tsx prisma/seed.ts"
  }
}
```

## 完整的 Seed 脚本

```typescript
// prisma/seed.ts
import { PrismaClient, BookStatus, UserRole } from '@prisma/client';
import * as bcrypt from 'bcrypt';

const prisma = new PrismaClient();

async function main() {
  console.log('🌱 开始初始化种子数据...\n');

  // ==================== 清理 ====================
  console.log('清理旧数据...');
  await prisma.bookTag.deleteMany();
  await prisma.tag.deleteMany();
  await prisma.borrowRecord.deleteMany();
  await prisma.book.deleteMany();
  await prisma.category.deleteMany();
  await prisma.user.deleteMany();

  // ==================== 用户 ====================
  console.log('创建用户...');
  const hashedPassword = await bcrypt.hash('123456', 10);

  const admin = await prisma.user.create({
    data: {
      username: 'admin',
      password: hashedPassword,
      nickname: '系统管理员',
      role: UserRole.ADMIN,
    },
  });

  const alice = await prisma.user.create({
    data: {
      username: 'alice',
      password: hashedPassword,
      nickname: 'Alice',
      role: UserRole.USER,
    },
  });

  const bob = await prisma.user.create({
    data: {
      username: 'bob',
      password: hashedPassword,
      nickname: 'Bob',
      role: UserRole.USER,
    },
  });

  console.log(`  ✅ 创建了 3 个用户`);

  // ==================== 分类 ====================
  console.log('创建图书分类...');
  const cs = await prisma.category.create({ data: { name: '计算机科学' } });
  const lit = await prisma.category.create({ data: { name: '文学小说' } });
  const hist = await prisma.category.create({ data: { name: '历史哲学' } });
  const sci = await prisma.category.create({ data: { name: '科学技术' } });
  const biz = await prisma.category.create({ data: { name: '经济管理' } });

  // 子分类
  const backend = await prisma.category.create({
    data: { name: '后端开发', parentId: cs.id },
  });
  const frontend = await prisma.category.create({
    data: { name: '前端开发', parentId: cs.id },
  });
  const novel = await prisma.category.create({
    data: { name: '小说', parentId: lit.id },
  });
  const essay = await prisma.category.create({
    data: { name: '散文', parentId: lit.id },
  });

  console.log(`  ✅ 创建了 9 个分类（含子分类）`);

  // ==================== 图书 ====================
  console.log('创建图书...');
  const booksData = [
    { title: 'NestJS 从入门到实践', author: '张三', isbn: '978-7-111-10001', price: 79.00, status: BookStatus.AVAILABLE, categoryId: backend.id },
    { title: 'Prisma ORM 权威指南', author: '李四', isbn: '978-7-111-10002', price: 89.00, status: BookStatus.AVAILABLE, categoryId: backend.id },
    { title: '深入理解 TypeScript', author: '王五', isbn: '978-7-111-10003', price: 69.00, status: BookStatus.BORROWED, categoryId: frontend.id },
    { title: 'Vue.js 设计与实现', author: '霍春阳', isbn: '978-7-111-10004', price: 95.00, status: BookStatus.AVAILABLE, categoryId: frontend.id },
    { title: '三体', author: '刘慈欣', isbn: '978-7-111-10005', price: 45.00, status: BookStatus.BORROWED, categoryId: novel.id },
    { title: '活着', author: '余华', isbn: '978-7-111-10006', price: 28.00, status: BookStatus.AVAILABLE, categoryId: novel.id },
    { title: '人类简史', author: '尤瓦尔·赫拉利', isbn: '978-7-111-10007', price: 68.00, status: BookStatus.AVAILABLE, categoryId: hist.id },
    { title: '时间简史', author: '霍金', isbn: '978-7-111-10008', price: 42.00, status: BookStatus.AVAILABLE, categoryId: sci.id },
    { title: '经济学原理', author: '曼昆', isbn: '978-7-111-10009', price: 88.00, status: BookStatus.AVAILABLE, categoryId: biz.id },
    { title: '百年孤独', author: '马尔克斯', isbn: '978-7-111-10010', price: 39.50, status: BookStatus.AVAILABLE, categoryId: novel.id },
    { title: '朝花夕拾', author: '鲁迅', isbn: '978-7-111-10011', price: 22.00, status: BookStatus.AVAILABLE, categoryId: essay.id },
    { title: 'Node.js 实战', author: '赵六', isbn: '978-7-111-10012', price: 75.00, status: BookStatus.MAINTENANCE, categoryId: backend.id },
  ];

  for (const data of booksData) {
    await prisma.book.create({
      data: {
        title: data.title,
        author: data.author,
        isbn: data.isbn,
        price: data.price,
        status: data.status,
        categoryId: data.categoryId,
      },
    });
  }

  console.log(`  ✅ 创建了 ${booksData.length} 本图书`);

  // ==================== 标签 ====================
  console.log('创建标签...');
  const tagNest = await prisma.tag.create({ data: { name: 'NestJS' } });
  const tagTS = await prisma.tag.create({ data: { name: 'TypeScript' } });
  const tagVue = await prisma.tag.create({ data: { name: 'Vue.js' } });
  const tagSciFi = await prisma.tag.create({ data: { name: '科幻' } });
  const tagLit = await prisma.tag.create({ data: { name: '文学经典' } });
  const tagEcon = await prisma.tag.create({ data: { name: '经济学' } });

  console.log(`  ✅ 创建了 6 个标签`);

  // ==================== 图书-标签关联 ====================
  const allBooks = await prisma.book.findMany();

  // 为前几本书添加标签
  const tagMappings: Record<string, number[]> = {
    'NestJS 从入门到实践': [tagNest.id, tagTS.id],
    'Prisma ORM 权威指南': [tagTS.id],
    'Vue.js 设计与实现': [tagVue.id, tagTS.id],
    '三体': [tagSciFi.id],
    '活着': [tagLit.id],
    '经济学原理': [tagEcon.id],
  };

  for (const book of allBooks) {
    const tagIds = tagMappings[book.title];
    if (tagIds) {
      for (const tagId of tagIds) {
        await prisma.bookTag.create({
          data: { bookId: book.id, tagId },
        });
      }
    }
  }

  console.log(`  ✅ 创建了图书-标签关联`);

  // ==================== 借阅记录 ====================
  console.log('创建借阅记录...');
  const borrowedBooks = await prisma.book.findMany({
    where: { status: BookStatus.BORROWED },
  });

  for (const book of borrowedBooks) {
    await prisma.borrowRecord.create({
      data: {
        userId: alice.id,
        bookId: book.id,
        borrowedAt: new Date(Date.now() - 7 * 24 * 60 * 60 * 1000), // 7 天前
      },
    });
  }

  // 更新借阅用户的 borrowCount
  await prisma.user.update({
    where: { id: alice.id },
    data: { borrowCount: borrowedBooks.length },
  });

  console.log(`  ✅ 创建了 ${borrowedBooks.length} 条借阅记录`);
  console.log('\n🎉 种子数据初始化完成！');
  console.log(`\n📋 测试账号：`);
  console.log(`   管理员: admin / 123456`);
  console.log(`   用户1: alice / 123456（已借阅 ${borrowedBooks.length} 本书）`);
  console.log(`   用户2: bob   / 123456`);
}

main()
  .catch((e) => {
    console.error('❌ Seed 失败:', e);
    process.exit(1);
  })
  .finally(async () => {
    await prisma.$disconnect();
  });
```

## 运行 Seed

```bash
# 单独运行
npx prisma db seed

# 或配合 migrate reset（清空数据库 + 重新建表 + 运行 seed）
npx prisma migrate reset

# 输出：
```
