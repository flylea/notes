# 图书管理系统实战场景

## 场景 1：定时清理过期借阅记录

```typescript
// src/tasks/overdue-borrow.task.ts
@Injectable()
export class OverdueBorrowTask {
  private readonly logger = new Logger(OverdueBorrowTask.name);

  constructor(private prisma: PrismaService) {}

  // 每天上午 10 点检查逾期未还的图书
  @Cron('0 0 10 * * *', { name: 'checkOverdue', timeZone: 'Asia/Shanghai' })
  async checkOverdueBorrows() {
    const overdueDays = 30; // 借阅超过 30 天视为逾期
    const threshold = new Date();
    threshold.setDate(threshold.getDate() - overdueDays);

    const overdueRecords = await this.prisma.borrowRecord.findMany({
      where: {
        returnedAt: null,
        borrowedAt: { lt: threshold },
        // 30 天前借出且仍未归还
      },
      include: {
        user: { select: { id: true, username: true } },
        book: { select: { id: true, title: true } },
      },
    });

    for (const record of overdueRecords) {
      this.logger.warn(
        `借阅逾期: 用户 ${record.user.username} 的《${record.book.title}》` +
        `已超期 ${Math.floor((Date.now() - record.borrowedAt.getTime()) / 86400000) - overdueDays} 天`,
      );
    }

    if (overdueRecords.length > 0) {
      this.logger.log(`共有 ${overdueRecords.length} 条逾期记录`);
    }
  }
}
```

## 场景 2：定时同步图书阅读统计

```typescript
// src/tasks/book-stats-task.ts
@Injectable()
export class BookStatsTask {
  constructor(
    private prisma: PrismaService,
    private eventEmitter: EventEmitter2,
  ) {}

  // 每小时更新一次图书热门榜单
  @Cron('0 0 * * * *', { name: 'updateHotBooks' })
  async updateHotBooks() {
    // 统计过去 7 天借阅最多的图书
    const weekAgo = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000);

    const hotBooks = await this.prisma.borrowRecord.groupBy({
      by: ['bookId'],
      where: { borrowedAt: { gte: weekAgo } },
      _count: { bookId: true },
      orderBy: { _count: { bookId: 'desc' } },
      take: 20,
    });

    // 发送事件——缓存系统对此感兴趣
    this.eventEmitter.emit('stats.hotBooksUpdated', {
      date: new Date().toISOString(),
      books: hotBooks.map(b => ({ bookId: b.bookId, borrowCount: b._count.bookId })),
    });
  }
}
```

## 场景 4：应用启动时预热

```typescript
@Injectable()
export class WarmupService implements OnApplicationBootstrap {
  constructor(
    private prisma: PrismaService,
    private redis: RedisService,
  ) {}

  async onApplicationBootstrap() {
    // 载入热门图书到 Redis
    const hotBooks = await this.prisma.book.findMany({
      orderBy: { borrows: { _count: 'desc' } },
      take: 50,
    });

    await this.redis.set(
      'cache:hotBooks',
      JSON.stringify(hotBooks),
      3600,
    );

    this.logger.log('应用预热完成：热门图书已缓存');
  }
}
```
