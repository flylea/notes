# Interceptor + Winston 自动请求日志

## 目标

自动记录每个 HTTP 请求的关键信息：

```
2024-01-15 10:30:05 INFO GET /book/list?keyword=NestJS 200 45ms - 192.168.1.1
2024-01-15 10:30:07 INFO POST /borrow/borrow 201 320ms - 192.168.1.2 userId=2
2024-01-15 10:30:08 WARN POST /user/login 401 12ms - 192.168.1.3 (failed login attempt)
```

## 实现 RequestLoggingInterceptor

```typescript
// src/common/interceptors/request-logging.interceptor.ts
import {
  Injectable,
  NestInterceptor,
  ExecutionContext,
  CallHandler,
  Logger,
} from '@nestjs/common';
import { Observable } from 'rxjs';
import { tap } from 'rxjs/operators';
import { Request, Response } from 'express';

@Injectable()
export class RequestLoggingInterceptor implements NestInterceptor {
  private readonly logger = new Logger('HTTP');

  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest<Request>();
    const { method, url, ip, body, query } = request;
    const userAgent = request.headers['user-agent'] || '';
    const startTime = Date.now();

    // 提取用户信息（如果有认证）
    const userId = (request as any).user?.sub;
    const username = (request as any).user?.username;

    return next.handle().pipe(
      tap({
        next: () => {
          const response = context.switchToHttp().getResponse<Response>();
          const duration = Date.now() - startTime;

          this.logger.log({
            method,
            url,
            statusCode: response.statusCode,
            duration: `${duration}ms`,
            ip: this.getRealIp(request),
            userAgent: userAgent.slice(0, 100),  // 截断长 user-agent
            userId,
            username,
            timestamp: new Date().toISOString(),
          });
        },
        error: (error) => {
          const duration = Date.now() - startTime;

          this.logger.warn({
            method,
            url,
            statusCode: error.status || 500,
            duration: `${duration}ms`,
            ip: this.getRealIp(request),
            error: error.message,
            userId,
            username,
            timestamp: new Date().toISOString(),
          });
        },
      }),
    );
  }

  private getRealIp(request: Request): string {
    return (
      (request.headers['x-forwarded-for'] as string)?.split(',')[0]?.trim() ||
      (request.headers['x-real-ip'] as string) ||
      request.ip ||
      'unknown'
    );
  }
}
```

## 注册全局拦截器

```typescript
// src/main.ts
import { RequestLoggingInterceptor } from './common/interceptors/request-logging.interceptor';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  // Winston Logger
  app.useLogger(app.get(WINSTON_MODULE_NEST_PROVIDER));

  // 请求日志拦截器
  app.useGlobalInterceptors(new RequestLoggingInterceptor());

  await app.listen(3000);
}
```

> 注意顺序：先设置 Winston Logger，再注册 Interceptor。因为 Interceptor 中使用 `new Logger('HTTP')` 依赖 Winston。

## 按状态码区分日志级别

```typescript
// 改进版：区分状态码
tap({
  next: () => {
    const statusCode = response.statusCode;
    const duration = Date.now() - startTime;

    const logData = {
      method,
      url,
      statusCode,
      duration: `${duration}ms`,
      ip: this.getRealIp(request),
    };

    if (statusCode >= 500) {
      this.logger.error('服务器错误', logData);
    } else if (statusCode >= 400) {
      this.logger.warn('客户端错误', logData);
    } else if (duration > 1000) {
      this.logger.warn('慢请求', logData);  // 超过 1 秒标记为慢请求
    } else {
      this.logger.log('请求完成', logData);
    }
  },
})
```

## 生产环境——排除健康检查

```typescript
intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
  const request = context.switchToHttp().getRequest<Request>();

  // 排除健康检查和静态文件
  const ignoredPaths = ['/health', '/favicon.ico', '/metrics'];
  if (ignoredPaths.includes(request.url)) {
    return next.handle();  // 不记录
  }

  // ... 正常日志逻辑
}
```

## 敏感数据脱敏

```typescript
// 不在日志中记录密码等敏感字段
private sanitizeBody(body: any): any {
  if (!body) return {};

  const sanitized = { ...body };
  const sensitiveFields = ['password', 'passwordConfirm', 'token', 'secret'];

  for (const field of sensitiveFields) {
    if (sanitized[field]) {
      sanitized[field] = '***';
    }
  }

  return sanitized;
}
```

> 请求 body 可能很大（如图书描述字段），生产环境通常不记录完整 body 或截断处理。

## 日志查询示例（生产环境）

```bash
# 查看今天的错误日志
grep '"level":"error"' logs/app-$(date +%Y-%m-%d).log | jq .

# 查看某个接口的响应时间分布
cat logs/app-$(date +%Y-%m-%d).log | jq 'select(.url=="/book/list") | .duration'

# 统计 HTTP 状态码分布
cat logs/app-$(date +%Y-%m-%d).log | jq -r '.statusCode' | sort | uniq -c | sort -rn

# 查看某用户的全部操作
grep '"userId":2' logs/app-$(date +%Y-%m-%d).log | jq '{method, url, timestamp}'
```
