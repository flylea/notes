## 1. 权限装饰器套件

```typescript
// src/common/decorators/auth.decorator.ts
import { applyDecorators, SetMetadata, UseGuards } from '@nestjs/common';
import { AuthGuard } from '../guards/auth.guard';
import { RolesGuard } from '../guards/roles.guard';
import { PermissionGuard } from '../guards/permission.guard';
import { ApiBearerAuth, ApiUnauthorizedResponse, ApiForbiddenResponse } from '@nestjs/swagger';

export const IS_PUBLIC_KEY = 'isPublic';
export const IsPublic = () => SetMetadata(IS_PUBLIC_KEY, true);

export const ROLES_KEY = 'roles';
export const PERMISSIONS_KEY = 'permissions';

export function RequireLogin() {
  return applyDecorators(
    UseGuards(AuthGuard),
    ApiBearerAuth(),
    ApiUnauthorizedResponse({ description: '未登录' }),
  );
}

export function RequireRoles(...roles: string[]) {
  return applyDecorators(
    UseGuards(AuthGuard, RolesGuard),
    SetMetadata(ROLES_KEY, roles),
    ApiBearerAuth(),
  );
}

export function RequirePermissions(...permissions: string[]) {
  return applyDecorators(
    UseGuards(AuthGuard, PermissionGuard),
    SetMetadata(PERMISSIONS_KEY, permissions),
    ApiBearerAuth(),
  );
}

export function AdminOnly() {
  return applyDecorators(
    UseGuards(AuthGuard, RolesGuard, PermissionGuard),
    SetMetadata(ROLES_KEY, ['admin']),
    SetMetadata(PERMISSIONS_KEY, ['*']),
    ApiBearerAuth(),
  );
}
```

## 2. 参数提取装饰器

```typescript
// src/common/decorators/param.decorator.ts
import { createParamDecorator, ExecutionContext, UnauthorizedException } from '@nestjs/common';

export interface JwtPayload {
  sub: string;      // user id
  username: string;
  roles: string[];
}

// 当前用户
export const CurrentUser = createParamDecorator(
  (data: keyof JwtPayload | undefined, ctx: ExecutionContext): JwtPayload | any => {
    const request = ctx.switchToHttp().getRequest();
    const user = request.user as JwtPayload;

    if (!user) {
      throw new UnauthorizedException('用户未登录');
    }

    return data ? user[data] : user;
  },
);

// 即可
// @CurrentUser() user: JwtPayload
// @CurrentUser('username') username: string
// @CurrentUser('sub') userId: string

// 客户端真实 IP
export const RealIp = createParamDecorator(
  (data: unknown, ctx: ExecutionContext): string => {
    const request = ctx.switchToHttp().getRequest();
    const xff = request.headers['x-forwarded-for'];
    if (xff) return xff.split(',')[0].trim();
    return request.ip || request.connection?.remoteAddress;
  },
);

// 用户角色列表
export const CurrentRoles = createParamDecorator(
  (data: unknown, ctx: ExecutionContext): string[] => {
    const request = ctx.switchToHttp().getRequest();
    return request.user?.roles || [];
  },
);
```

## 4. 在 Controller 中完整使用

```typescript
// src/book/book.controller.ts
import {
  Controller, Get, Post, Body, Query,
} from '@nestjs/common';
import {
  IsPublic, RequireLogin, RequireRoles, AdminOnly,
} from '../common/decorators/auth.decorator';
import { CurrentUser } from '../common/decorators/param.decorator';

@Controller('book')
export class BookController {
  // 公开接口——任何人都能看图书列表
  @Get('list')
  @IsPublic()
  list(@Query() query: QueryBookDto) {
    return this.bookService.list(query);
  }

  // 需要登录才能借书
  @Post('borrow')
  @RequireLogin()
  borrow(@Query('id') id: string, @CurrentUser('sub') userId: string) {
    // userId 自动从 JWT 中提取
    return this.borrowService.borrowBook(userId, id);
  }

  // 需要特定角色才能创建图书
  @Post('create')
  @RequireRoles('admin', 'librarian')
  create(@Body() dto: CreateBookDto, @CurrentUser() user: JwtPayload) {
    console.log(`图书由 ${user.username} 创建`);
    return this.bookService.create(dto);
  }

  // 管理员专用
  @Post('delete')
  @AdminOnly()
  delete(@Query('id') id: string) {
    return this.bookService.remove(id);
  }
}
```

## 装饰器设计的 SOLID 原则

1. **单一职责**：每个装饰器只做一件事（`@IsPublic` 只标记公开，`@RequireRoles` 只标记角色）
2. **开放封闭**：新增权限策略只需新增组合装饰器，不改动现有代码
3. **组合胜过继承**：`@AdminOnly` = 组合 3 个 Guard + 2 个 SetMetadata，不是继承
4. **声明式 > 命令式**：`@RequireRoles('admin')` 比在方法体里 `if (!user.roles.includes('admin')) throw ...` 清晰 10 倍

> 自定义装饰器是 Nest 进阶的分水岭。能从 CRUD 提升到"编写声明式业务语义"——你的团队会感谢你。
