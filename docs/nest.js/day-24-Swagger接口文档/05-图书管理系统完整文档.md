# 为图书管理系统添加完整 Swagger 文档

## 改造后的 BookController

```typescript
// src/book/book.controller.ts
import { Controller, Get, Post, Body, Query } from '@nestjs/common';
import { ApiTags, ApiOperation, ApiQuery, ApiBody, ApiResponse, ApiBearerAuth } from '@nestjs/swagger';
import { BookService } from './book.service';
import { CurrentUser } from '../auth/user.decorator';
import { JwtPayload } from '../auth/auth.guard';

@Controller('book')
@ApiTags('图书管理')
export class BookController {
  constructor(private readonly bookService: BookService) {}

  @Get('list')
  @ApiOperation({ summary: '查询图书列表', description: '分页查询图书，支持按书名/作者/ISBN 搜索' })
  @ApiQuery({ name: 'page', required: false, type: Number, example: 1 })
  @ApiQuery({ name: 'size', required: false, type: Number, example: 10 })
  @ApiQuery({ name: 'keyword', required: false, type: String, description: '搜索关键词' })
  @ApiResponse({ status: 200, description: '查询成功' })
  async list(
    @Query('page') page?: string,
    @Query('size') size?: string,
    @Query('keyword') keyword?: string,
  ) {
    return this.bookService.list({
      page: page ? parseInt(page, 10) : 1,
      size: size ? parseInt(size, 10) : 10,
      keyword,
    });
  }

  @Get('detail')
  @ApiOperation({ summary: '查看图书详情' })
  @ApiQuery({ name: 'id', required: true, type: Number, example: 1 })
  @ApiResponse({ status: 200, description: '查询成功' })
  @ApiResponse({ status: 400, description: '图书不存在' })
  async detail(@Query('id') id: string) {
    return this.bookService.findById(parseInt(id, 10));
  }

  @Post('create')
  @ApiBearerAuth('bearer')
  @ApiOperation({ summary: '新增图书', description: '需要 book:create 权限' })
  @ApiBody({
    schema: {
      type: 'object',
      properties: {
        title: { type: 'string', example: 'NestJS 实战' },
        author: { type: 'string', example: '张三' },
        isbn: { type: 'string', example: '978-7-111-10001' },
        price: { type: 'number', example: 79.0 },
        description: { type: 'string', example: '一本好书' },
        categoryId: { type: 'number', example: 1 },
      },
    },
  })
  @ApiResponse({ status: 201, description: '创建成功' })
  @ApiResponse({ status: 400, description: '参数校验失败' })
  @ApiResponse({ status: 401, description: '未登录' })
  @ApiResponse({ status: 403, description: '权限不足' })
  async create(
    @Body() body: any,
    @CurrentUser() user: JwtPayload,
  ) {
    return this.bookService.create(body);
  }

  @Post('update')
  @ApiBearerAuth('bearer')
  @ApiOperation({ summary: '更新图书', description: '需要 book:update 权限' })
  @ApiBody({
    schema: {
      type: 'object',
      required: ['id'],
      properties: {
        id: { type: 'number', example: 1 },
        title: { type: 'string', example: 'NestJS 实战（修订版）' },
        author: { type: 'string' },
        price: { type: 'number', example: 89.0 },
      },
    },
  })
  @ApiResponse({ status: 200, description: '更新成功' })
  @ApiResponse({ status: 400, description: '图书不存在' })
  @ApiResponse({ status: 401, description: '未登录' })
  async update(@Body('id') id: string, @Body() body: any) {
    return this.bookService.update(parseInt(id, 10), body);
  }

  @Post('delete')
  @ApiBearerAuth('bearer')
  @ApiOperation({ summary: '删除图书', description: '需要 book:delete 权限' })
  @ApiBody({ schema: { properties: { id: { type: 'number', example: 1 } } } })
  @ApiResponse({ status: 200, description: '删除成功' })
  @ApiResponse({ status: 400, description: '图书不存在' })
  @ApiResponse({ status: 401, description: '未登录' })
  @ApiResponse({ status: 403, description: '权限不足' })
  async delete(@Body('id') id: string) {
    return this.bookService.remove(parseInt(id, 10));
  }
}
```

## 改造后的 BorrowController

```typescript
// src/borrow/borrow.controller.ts
import { Controller, Get, Post, Body, Query } from '@nestjs/common';
import { ApiTags, ApiOperation, ApiBody, ApiQuery, ApiResponse, ApiBearerAuth } from '@nestjs/swagger';
import { BorrowService } from './borrow.service';
import { CurrentUser } from '../auth/user.decorator';
import { JwtPayload } from '../auth/auth.guard';

@Controller('borrow')
@ApiTags('借阅管理')
@ApiBearerAuth('bearer')     // 整个 Controller 都需要 Token
export class BorrowController {
  constructor(private readonly borrowService: BorrowService) {}

  @Post('borrow')
  @ApiOperation({ summary: '借书', description: '借阅一本图书，每人最多同时借 3 本' })
  @ApiBody({
    schema: {
      type: 'object',
      required: ['bookId'],
      properties: {
        bookId: { type: 'number', example: 1, description: '图书 ID' },
      },
    },
  })
  @ApiResponse({ status: 201, description: '借阅成功' })
  @ApiResponse({ status: 400, description: '图书已被借出/超过上限/已借过' })
  @ApiResponse({ status: 401, description: '未登录' })
  async borrow(
    @Body('bookId') bookId: string,
    @CurrentUser('sub') userId: number,
  ) {
    return this.borrowService.borrowBook(userId, parseInt(bookId, 10));
  }

  @Post('return')
  @ApiOperation({ summary: '还书' })
  @ApiBody({
    schema: {
      type: 'object',
      required: ['bookId'],
      properties: {
        bookId: { type: 'number', example: 1 },
      },
    },
  })
  @ApiResponse({ status: 200, description: '归还成功' })
  @ApiResponse({ status: 400, description: '未找到借阅记录' })
  async returnBook(
    @Body('bookId') bookId: string,
    @CurrentUser('sub') userId: number,
  ) {
    return this.borrowService.returnBook(userId, parseInt(bookId, 10));
  }

  @Get('my-borrows')
  @ApiOperation({ summary: '我的当前借阅', description: '查询当前用户未归还的图书' })
  @ApiResponse({ status: 200, description: '查询成功' })
  async myBorrows(@CurrentUser('sub') userId: number) {
    return this.borrowService.getMyBorrows(userId);
  }

  @Get('history')
  @ApiOperation({ summary: '借阅历史' })
  @ApiQuery({ name: 'page', required: false, type: Number, example: 1 })
  @ApiQuery({ name: 'size', required: false, type: Number, example: 10 })
  @ApiResponse({ status: 200, description: '查询成功' })
  async history(
    @CurrentUser('sub') userId: number,
    @Query('page') page?: string,
    @Query('size') size?: string,
  ) {
    return this.borrowService.getBorrowHistory(userId, {
      page: page ? parseInt(page, 10) : 1,
      size: size ? parseInt(size, 10) : 10,
    });
  }
}
```

## 用 @ApiBearerAuth 标记公开接口

Swagger UI 中，标记了 `@ApiBearerAuth('bearer')` 的接口会自动在请求中添加 `Authorization: Bearer <token>` 头。用户先在右上角输入 Token，然后 Try it out 时自动携带。

公开接口不需要这个装饰器：

```typescript
@Controller('user')
@ApiTags('用户认证')
export class UserController {
  @Post('register')
  @ApiOperation({ summary: '用户注册' })
  async register() {}   // 不需要 @ApiBearerAuth

  @Post('login')
  @ApiOperation({ summary: '用户登录' })
  async login() {}      // 不需要 @ApiBearerAuth
}
```
