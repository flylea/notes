# 关注关系 + 排行榜实战

## 场景 1：读者关注关系（Set 实现）

```typescript
// src/social/follow.service.ts
@Injectable()
export class FollowService {
  constructor(private redis: RedisService) {}

  // 用户 A 关注用户 B
  async follow(followerId: number, followingId: number) {
    // followerId 关注了谁
    await this.redis.sadd(`user:${followerId}:following`, String(followingId));
    // followingId 被谁关注
    await this.redis.sadd(`user:${followingId}:followers`, String(followerId));
  }

  // 取消关注
  async unfollow(followerId: number, followingId: number) {
    await this.redis.sadd(`user:${followerId}:following`, String(followingId));
    // ...
  }

  // 查看关注列表
  async getFollowing(userId: number): Promise<number[]> {
    const members = await this.redis.smembers(`user:${userId}:following`);
    return members.map(Number);
  }

  // 共同关注——Set 交集
  async getCommonFollowing(userA: number, userB: number): Promise<number[]> {
    // 使用 SINTER 而不是在应用层计算
    const pipeline = await this.redis.executePipeline([
      ['sinter', `user:${userA}:following`, `user:${userB}:following`],
    ]);
    const [err, result] = pipeline[0];
    return (result as string[]).map(Number);
  }

  // 推荐关注：A 关注的人也在关注谁（二度关注）
  async suggestFollows(userId: number, limit = 10) {
    const myFollowing = await this.redis.smembers(`user:${userId}:following`);

    // 收集所有二度用户
    const candidates = new Map<number, number>(); // userId → 共同关注数

    for (const fid of myFollowing) {
      const theirFollowing = await this.redis.smembers(`user:${fid}:following`);
      for (const tfid of theirFollowing) {
        const tid = Number(tfid);
        if (tid !== userId && !myFollowing.includes(tfid)) {
          candidates.set(tid, (candidates.get(tid) || 0) + 1);
        }
      }
    }

    // 按共同关注数降序排列
    return Array.from(candidates.entries())
      .sort((a, b) => b[1] - a[1])
      .slice(0, limit)
      .map(([userId, count]) => ({ userId, commonCount: count }));
  }
}
```

## 场景 2：图书借阅排行榜（ZSet 实现）

```typescript
// src/stats/leaderboard.service.ts
@Injectable()
export class LeaderboardService {
  constructor(private redis: RedisService, private prisma: PrismaService) {}

  // 记录借阅（ZSet 分数 +1）
  async recordBorrow(bookId: number) {
    await this.redis.zincrby('leaderboard:borrow:weekly', 1, `book:${bookId}`);
    await this.redis.zincrby('leaderboard:borrow:total', 1, `book:${bookId}`);
  }

  // 本周热门 Top N
  async getWeeklyTop(limit = 10) {
    const results = await this.redis.zrevrange(
      'leaderboard:borrow:weekly',
      0,
      limit - 1,
      true,
    );

    // results: ["book:3", "15", "book:1", "12", ...]
    return this.formatResults(results);
  }

  // 总榜 Top N
  async getTotalTop(limit = 10) {
    const results = await this.redis.zrevrange(
      'leaderboard:borrow:total',
      0,
      limit - 1,
      true,
    );
    return this.formatResults(results);
  }

  // 查询某本书的排名
  async getBookRank(bookId: number) {
    const [weeklyRank, weeklyScore, totalRank, totalScore] = await Promise.all([
      this.redis.zrevrank('leaderboard:borrow:weekly', `book:${bookId}`),
      this.redis.zscore('leaderboard:borrow:weekly', `book:${bookId}`),
      this.redis.zrevrank('leaderboard:borrow:total', `book:${bookId}`),
      this.redis.zscore('leaderboard:borrow:total', `book:${bookId}`),
    ]);

    return {
      weekly: weeklyRank !== null ? { rank: weeklyRank + 1, count: Number(weeklyScore) } : null,
      total: totalRank !== null ? { rank: totalRank + 1, count: Number(totalScore) } : null,
    };
  }

  // 每周一重置周榜
  @Cron('0 0 0 * * 1')  // 每周一凌晨
  async resetWeeklyLeaderboard() {
    await this.redis.del('leaderboard:borrow:weekly');
  }

  private async formatResults(results: string[]) {
    const formatted: Array<{ bookId: number; count: number }> = [];
    for (let i = 0; i < results.length; i += 2) {
      formatted.push({
        bookId: parseInt(results[i].split(':')[1], 10),
        count: parseInt(results[i + 1], 10),
      });
    }

    // 补充图书基本信息
    const bookIds = formatted.map(f => f.bookId);
    const books = await this.prisma.book.findMany({
      where: { id: { in: bookIds } },
      select: { id: true, title: true, author: true, cover: true },
    });

    const bookMap = new Map(books.map(b => [b.id, b]));
    return formatted.map(f => ({
      ...f,
      book: bookMap.get(f.bookId),
    }));
  }
}
```
