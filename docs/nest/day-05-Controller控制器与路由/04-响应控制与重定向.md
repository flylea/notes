# 响应控制：@HttpCode / @Header / @Redirect

## @HttpCode — 自定义 HTTP 状态码

```typescript
@Post()
@HttpCode(201)  // 默认 POST 返回 201，不指定的话是 200
create(@Body() dto: CreateUserDto) {
  return this.userService.create(dto);
}

@Get()
@HttpCode(200)  // GET 默认就是 200，可以省略
findAll() {
  return this.userService.findAll();
}
```

Nest 默认的行为：
- POST → 201（Created）
- GET → 200（OK）
- 其他 → 200

```typescript
@Post('register')
@HttpCode(201)  // 显式声明：创建成功
register() {}

@Post('login')
@HttpCode(200)  // 登录成功（不是创建资源）
login() {}
```

## @Header — 自定义响应头

```typescript
@Get()
@Header('Cache-Control', 'max-age=3600')  // 缓存 1 小时
findAll() {
  return this.userService.findAll();
}

// 多个响应头
@Get()
@Header('X-Total-Count', '100')
@Header('X-Custom', 'value')
findAll() {}
```

## @Redirect — 重定向

```typescript
@Get()
@Redirect('https://docs.nestjs.com', 302)  // 默认值，可省略
redirect() {
  // 静态重定向
}

@Get('short/:code')
@Redirect()
async shortLink(@Param('code') code: string) {
  const url = await this.shortLinkService.resolve(code);
  return { url };  // 动态重定向：返回的 { url } 中的 url 值即为重定向目标
}
```

> `@Redirect()` 返回 `{ url: '...', statusCode: 302 }` 实现动态重定向。

## return 的自动序列化

Nest 会自动将 Controller 方法的返回值序列化为 JSON：

```typescript
@Get()
findAll() {
  return this.userService.findAll();  // 返回 Promise<User[]>
}
// 浏览器收到: [{ "id": 1, "username": "john" }, ...]
// 自动设置 Content-Type: application/json

@Get()
findOne() {
  return 'hello';  // 返回 string
}
// 浏览器收到: "hello"

@Get()
findAll() {
  return { data: [], code: 0, message: 'ok' };  // 返回对象
}
// 浏览器收到: { "data": [], "code": 0, "message": "ok" }
```

> 序列化是**递归的**——嵌套对象、数组、Date 都会正确处理。不需要 `JSON.stringify()`。

## 统一响应格式的最佳实践

不要在每个 Controller 中手动包装响应，用 Interceptor（Day 10 会详细介绍）：

```typescript
// 在 Interceptor 中统一包装
@Injectable()
export class ResponseInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    return next.handle().pipe(
      map(data => ({
        code: 0,
        data,
        message: 'ok',
      })),
    );
  }
}

// Controller 只需要 return 数据本身
@Get()
findAll() {
  return this.userService.findAll();
  // Interceptor 自动包装为: { code: 0, data: [...], message: 'ok' }
}
```
