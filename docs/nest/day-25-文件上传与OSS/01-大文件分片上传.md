# 大文件分片上传与断点续传

## 为什么需要分片上传

```
上传一个 2GB 的视频文件：
  直接上传：传了 1.8GB → 网络断开 → 从头开始 ❌
  分片上传：切成 5MB 一片 → 传了 360 片 → 网络断开 → 只重传最后一片 ✅
```

## 原理

```
┌────────────────────────────────────────────────────┐
│  1. 前端用 Blob.slice 将文件切成 N 片                │
│     每片 5MB，计算每片的 MD5                         │
│                                                     │
│  2. 上传前调用 /upload/init                         │
│     返回 uploadId + 已上传的分片列表                  │
│                                                     │
│  3. 跳过已上传的分片，只传缺失的分片                   │
│     POST /upload/chunk                              │
│     FormData: { chunk, uploadId, chunkIndex }       │
│                                                     │
│  4. 全部传完 → 调用 /upload/merge                   │
│     服务器合并所有分片 → 生成最终文件                 │
└────────────────────────────────────────────────────┘
```

## 后端实现

```typescript
// src/upload/chunk-upload.service.ts
import { Injectable } from '@nestjs/common';
import * as fs from 'fs';
import * as path from 'path';
import * as crypto from 'crypto';

interface UploadTask {
  uploadId: string;
  fileName: string;
  fileSize: number;
  totalChunks: number;
  chunkSize: number;
  uploadedChunks: number[];
  createdAt: Date;
}

@Injectable()
export class ChunkUploadService {
  private tasks = new Map<string, UploadTask>();
  private readonly tempDir = './uploads/temp';
  private readonly finalDir = './uploads/books';

  constructor() {
    fs.mkdirSync(this.tempDir, { recursive: true });
    fs.mkdirSync(this.finalDir, { recursive: true });
  }

  // 初始化上传任务
  initUpload(fileName: string, fileSize: number, chunkSize = 5 * 1024 * 1024) {
    const uploadId = crypto.randomUUID();
    const totalChunks = Math.ceil(fileSize / chunkSize);

    this.tasks.set(uploadId, {
      uploadId,
      fileName,
      fileSize,
      totalChunks,
      chunkSize,
      uploadedChunks: [],
      createdAt: new Date(),
    });

    return { uploadId, totalChunks, chunkSize };
  }

  // 上传分片
  async uploadChunk(
    uploadId: string,
    chunkIndex: number,
    chunkBuffer: Buffer,
  ) {
    const task = this.tasks.get(uploadId);
    if (!task) throw new Error('上传任务不存在');

    // 保存分片到临时目录
    const chunkPath = path.join(this.tempDir, `${uploadId}_${chunkIndex}`);
    await fs.promises.writeFile(chunkPath, chunkBuffer);

    // 记录已上传的分片
    task.uploadedChunks.push(chunkIndex);

    return {
      chunkIndex,
      uploaded: task.uploadedChunks.length,
      total: task.totalChunks,
    };
  }

  // 合并分片
  async mergeChunks(uploadId: string) {
    const task = this.tasks.get(uploadId);
    if (!task) throw new Error('上传任务不存在');

    if (task.uploadedChunks.length !== task.totalChunks) {
      throw new Error('还有分片未上传');
    }

    // 按序号排序后合并
    const finalPath = path.join(this.finalDir, task.fileName);
    const writeStream = fs.createWriteStream(finalPath);

    for (let i = 0; i < task.totalChunks; i++) {
      const chunkPath = path.join(this.tempDir, `${uploadId}_${i}`);
      const chunkData = await fs.promises.readFile(chunkPath);
      writeStream.write(chunkData);

      // 清理分片
      await fs.promises.unlink(chunkPath);
    }

    writeStream.end();

    // 清理任务记录
    this.tasks.delete(uploadId);

    return { url: `/uploads/books/${task.fileName}` };
  }

  // 检查上传进度（断点续传关键）
  getProgress(uploadId: string) {
    const task = this.tasks.get(uploadId);
    if (!task) throw new Error('上传任务不存在');

    return {
      uploadId: task.uploadId,
      uploadedChunks: task.uploadedChunks.sort((a, b) => a - b),
      totalChunks: task.totalChunks,
      progress: Math.round((task.uploadedChunks.length / task.totalChunks) * 100),
    };
  }
}
```

## Controller

```typescript
// src/upload/upload.controller.ts
import {
  Controller,
  Post,
  Get,
  Body,
  Query,
  UploadedFile,
  UseInterceptors,
} from '@nestjs/common';
import { FileInterceptor } from '@nestjs/platform-express';
import { ChunkUploadService } from './chunk-upload.service';

@Controller('upload')
export class UploadController {
  constructor(private readonly chunkUploadService: ChunkUploadService) {}

  @Post('init')
  async init(@Body() body: { fileName: string; fileSize: number }) {
    return this.chunkUploadService.initUpload(body.fileName, body.fileSize);
  }

  @Post('chunk')
  @UseInterceptors(FileInterceptor('chunk'))
  async chunk(
    @UploadedFile() file: Express.Multer.File,
    @Body() body: { uploadId: string; chunkIndex: string },
  ) {
    return this.chunkUploadService.uploadChunk(
      body.uploadId,
      parseInt(body.chunkIndex, 10),
      file.buffer,
    );
  }

  @Post('merge')
  async merge(@Body('uploadId') uploadId: string) {
    return this.chunkUploadService.mergeChunks(uploadId);
  }

  @Get('progress')
  async progress(@Query('uploadId') uploadId: string) {
    return this.chunkUploadService.getProgress(uploadId);
  }
}
```

## 前端分片上传代码（参考）

```typescript
// 前端分片上传
async function chunkUpload(file: File) {
  const CHUNK_SIZE = 5 * 1024 * 1024; // 5MB
  const totalChunks = Math.ceil(file.size / CHUNK_SIZE);

  // 1. 初始化
  const { uploadId } = await fetch('/upload/init', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ fileName: file.name, fileSize: file.size }),
  }).then(r => r.json());

  // 2. 上传分片
  for (let i = 0; i < totalChunks; i++) {
    const start = i * CHUNK_SIZE;
    const chunk = file.slice(start, start + CHUNK_SIZE);

    const formData = new FormData();
    formData.append('chunk', chunk);
    formData.append('uploadId', uploadId);
    formData.append('chunkIndex', String(i));

    await fetch('/upload/chunk', { method: 'POST', body: formData });
    console.log(`进度: ${Math.round(((i + 1) / totalChunks) * 100)}%`);
  }

  // 3. 合并
  const result = await fetch('/upload/merge', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ uploadId }),
  }).then(r => r.json());

  return result.url;
}
```
