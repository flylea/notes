# 新增 BorrowController 和 BorrowModule

## 创建 BorrowModule

```typescript
// src/borrow/borrow.module.ts
import { Module } from '@nestjs/common';
import { BorrowService } from './borrow.service';
import { BorrowController } from './borrow.controller';

@Module({
  controllers: [BorrowController],
  providers: [BorrowService],
  exports: [BorrowService],
})
export class BorrowModule {}
```

## 创建 BorrowController

```typescript
// src/borrow/borrow.controller.ts
import {
  Controller,
  Get,
  Post,
  Body,
  Query,
  BadRequestException,
} from '@nestjs/common';
import { BorrowService } from './borrow.service';

@Controller('borrow')
export class BorrowController {
  constructor(private readonly borrowService: BorrowService) {}

  // 借书
  @Post('borrow')
  async borrow(
    @Body('userId') userId: string,
    @Body('bookId') bookId: string,
  ) {
    const uid = parseInt(userId, 10);
    const bid = parseInt(bookId, 10);

    if (isNaN(uid) || isNaN(bid)) {
      throw new BadRequestException('参数格式错误');
    }

    return this.borrowService.borrowBook(uid, bid);
  }

  // 还书
  @Post('return')
  async returnBook(
    @Body('userId') userId: string,
    @Body('bookId') bookId: string,
  ) {
    const uid = parseInt(userId, 10);
    const bid = parseInt(bookId, 10);

    if (isNaN(uid) || isNaN(bid)) {
      throw new BadRequestException('参数格式错误');
    }

    return this.borrowService.returnBook(uid, bid);
  }

  // 我的当前借阅
  @Get('my-borrows')
  async myBorrows(@Query('userId') userId: string) {
    const uid = parseInt(userId, 10);
    if (isNaN(uid)) {
      throw new BadRequestException('参数格式错误');
    }
    return this.borrowService.getMyBorrows(uid);
  }

  // 借阅历史
  @Get('history')
  async history(
    @Query('userId') userId: string,
    @Query('page') page?: string,
    @Query('size') size?: string,
  ) {
    const uid = parseInt(userId, 10);
    if (isNaN(uid)) {
      throw new BadRequestException('参数格式错误');
    }

    return this.borrowService.getBorrowHistory(uid, {
      page: page ? parseInt(page, 10) : 1,
      size: size ? parseInt(size, 10) : 10,
    });
  }

  // 检查图书可借状态
  @Get('check')
  async check(@Query('bookId') bookId: string) {
    const bid = parseInt(bookId, 10);
    if (isNaN(bid)) {
      throw new BadRequestException('参数格式错误');
    }
    return this.borrowService.checkAvailability(bid);
  }
}
```

## 注册到 AppModule

```typescript

import { Module } from '@nestjs/common';

import { BorrowModule } from './borrow/borrow.module';  // ← 新增

  imports: [
    PrismaModule,
    UserModule,

```
