# createParamDecorator — 自定义参数装饰器

## 问题：重复提取用户信息

```typescript
// 每个接口都这样写
@Get('profile')
getProfile(@Req() req: Request) {
  const user = (req as any).user;  // 手动从 req 中提取
  return this.userService.getProfile(user.id);
}

@Post('update')
updateProfile(@Req() req: Request, @Body() dto: UpdateUserDto) {
  const user = (req as any).user;  // 又重复了
  return this.userService.update(user.id, dto);
}
```

Nest 的解：`createParamDecorator` — 一行获取当前用户。

## 基本用法

```typescript
import { createParamDecorator, ExecutionContext } from '@nestjs/common';

export const UserInfo = createParamDecorator(
  (data: unknown, ctx: ExecutionContext) => {
    const request = ctx.switchToHttp().getRequest();
    return request.user;  // 返回挂载在 request 上的用户信息
  },
);
```

### 使用

```typescript
@Get('profile')
getProfile(@UserInfo() user: any) {
  // user 就是 request.user
  console.log(user.id, user.username);
  return this.userService.getProfile(user.id);
}
```

## 提取特定字段

```typescript
// 支持 data 参数：@UserInfo('id') → 只返回 id
export const UserInfo = createParamDecorator(
  (data: string | undefined, ctx: ExecutionContext) => {
    const request = ctx.switchToHttp().getRequest();
    const user = request.user;

    // 如果没有传 data，返回整个 user 对象
    // 如果传了 data（如 'id'）→ 返回特定字段
    return data ? user?.[data] : user;
  },
);
```

```typescript
@Get('profile')
getProfile(
  @UserInfo('id') userId: string,
  @UserInfo('username') username: string,
) {
  // userId = request.user.id
  // username = request.user.username
}
```

## 实战：三个常用参数装饰器

### 1. @UserInfo — 获取当前用户

```typescript
export const UserInfo = createParamDecorator(
  (data: string | undefined, ctx: ExecutionContext) => {
    const request = ctx.switchToHttp().getRequest();
    const user = request.user;

    if (!user) {
      throw new UnauthorizedException('用户未登录');
    }

    return data ? user?.[data] : user;
  },
);
```

### 2. @ClientIp — 获取客户端真实 IP

```typescript
export const ClientIp = createParamDecorator(
  (data: unknown, ctx: ExecutionContext): string => {
    const request = ctx.switchToHttp().getRequest();

    // 优先取代理转发的真实 IP
    const forwarded = request.headers['x-forwarded-for'];
    if (forwarded) {
      return (Array.isArray(forwarded) ? forwarded[0] : forwarded).split(',')[0].trim();
    }

    return request.ip || request.connection?.remoteAddress;
  },
);
```

### 3. @Pagination — 提取分页参数

```typescript
export interface PaginationParams {
  page: number;
  size: number;
  skip: number;
}

export const Pagination = createParamDecorator(
  (data: unknown, ctx: ExecutionContext): PaginationParams => {
    const request = ctx.switchToHttp().getRequest();
    const page = parseInt(request.query.page, 10) || 1;
    const size = parseInt(request.query.size, 10) || 10;

    return {
      page: Math.max(1, page),
      size: Math.min(100, Math.max(1, size)),  // 限制在 1-100
      skip: (page - 1) * size,
    };
  },
);
```

```typescript
@Get('list')
list(@Pagination() pagination: PaginationParams) {
  // pagination = { page: 1, size: 10, skip: 0 }
  return this.bookService.list(pagination);
}
```

## createParamDecorator 的管道支持

可以和 `ValidationPipe` 配合使用：

```typescript
@Get('list')
list(
  @Pagination(ValidationPipe) pagination: PaginationParams,
) {
  // ValidationPipe 会校验 PaginationParams 的装饰器
}
```

## Express vs Fastify

`createParamDecorator` 是跨平台兼容的——通过 `ExecutionContext` 而非直接访问 `req`/`res`：

```typescript
// ✅ 兼容 Express 和 Fastify
export const UserInfo = createParamDecorator(
  (data: unknown, ctx: ExecutionContext) => {
    const request = ctx.switchToHttp().getRequest();
    return request.user;
  },
);

// ❌ 不兼容（直接用了 Express Request 类型）
export const UserInfo = (req: Request) => req.user;
```

> `createParamDecorator` 让代码从"手动从 req 提取"变成"声明式获取"——`@UserInfo()` 比 `@Req() + req.user` 更语义化，也更方便测试。
